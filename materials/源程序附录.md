"""VisDrone原始数据转换；不改动原始文件，训练遮罩与原图评估分离。"""
import csv
import hashlib
import shutil
import zipfile
import urllib.request
import urllib.error
from collections import Counter
from pathlib import Path
import numpy as np
from PIL import Image, ImageDraw
import yaml
from .runtime import write_json, read_json

SPLITS = {"train": "VisDrone2019-DET-train", "val": "VisDrone2019-DET-val", "test": "VisDrone2019-DET-test-dev"}
EXPECTED_COUNTS = {"train": 6471, "val": 548, "test": 1610}
BASE_URL = "https://github.com/ultralytics/assets/releases/download/v0.0.0/"


def sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for b in iter(lambda: f.read(1024 * 1024), b""):
            h.update(b)
    return h.hexdigest()


def download_dataset(raw, splits=("train", "val", "test"), offline=False):
    raw = Path(raw).resolve(); raw.mkdir(parents=True, exist_ok=True)
    sources = []
    for split in splits:
        folder = SPLITS[split]
        archive = raw / (folder + ".zip")
        extracted = raw / folder
        if not archive.exists():
            images = len(list((extracted / "images").glob("*.jpg")))
            labels = len(list((extracted / "annotations").glob("*.txt")))
            if images == EXPECTED_COUNTS[split] and labels == EXPECTED_COUNTS[split]:
                sources.append({"split": split, "source": "local_extracted", "images": images, "annotations": labels})
                continue
            if offline:
                raise FileNotFoundError(
                    f"离线模式缺少 {archive} 或完整的 {extracted}；"
                    f"请从可联网电脑复制 {archive.name} 到 data/raw，现有图片/标注 {images}/{labels}。"
                )
            temp = archive.with_suffix(".zip.part")
            print(f"下载 {BASE_URL + archive.name}", flush=True)
            try:
                urllib.request.urlretrieve(BASE_URL + archive.name, temp)
            except (urllib.error.URLError, OSError) as exc:
                raise RuntimeError(
                    f"下载 {archive.name} 失败：{exc}。可在能访问GitHub的电脑下载该ZIP，"
                    f"复制到 {raw}，再运行 run.py download --offline。"
                ) from exc
            temp.replace(archive)
        with zipfile.ZipFile(archive) as z:
            bad = z.testzip()
            if bad:
                raise ValueError(f"压缩包校验失败: {bad}")
            for member in z.infolist():
                dest = (raw / member.filename).resolve()
                if not dest.is_relative_to(raw):
                    raise ValueError("压缩包包含越界路径")
            have_images = len(list((extracted / "images").glob("*.jpg")))
            have_labels = len(list((extracted / "annotations").glob("*.txt")))
            if have_images != EXPECTED_COUNTS[split] or have_labels != EXPECTED_COUNTS[split]:
                z.extractall(raw)
        images = len(list((extracted / "images").glob("*.jpg")))
        labels = len(list((extracted / "annotations").glob("*.txt")))
        if images != EXPECTED_COUNTS[split] or labels != EXPECTED_COUNTS[split]:
            raise ValueError(f"{folder}解压不完整：图片/标注 {images}/{labels}，应各有{EXPECTED_COUNTS[split]}。")
        sources.append({"split": split, "url": BASE_URL + archive.name, "sha256": sha256(archive)})
    write_json(raw / "download_provenance.json", sources)


def parse_annotation(path, width, height):
    boxes, ignores, attributes, issues = [], [], [], Counter()
    for line in Path(path).read_text(encoding="utf-8-sig").splitlines():
        if not line.strip():
            continue
        fields = line.strip().rstrip(",").split(",")
        if len(fields) < 8:
            raise ValueError(f"错误标注格式: {path}: {line}")
        x, y, w, h, score, category, truncation, occlusion = map(float, fields[:8])
        if not np.isfinite([x, y, w, h, score, category, truncation, occlusion]).all():
            raise ValueError(f"非有限标注: {path}")
        issues[f"raw_class_{int(category)}"] += 1
        original = np.array([x, y, x+w, y+h])
        box = np.clip(original, [0, 0, 0, 0], [width, height, width, height]).tolist()
        if box[2] <= box[0] or box[3] <= box[1]:
            issues["invalid_area"] += 1
            continue
        if not np.array_equal(original, box):
            issues["clipped_boxes"] += 1
        if score == 0 or int(category) in (0, 11):
            ignores.append(box)
        elif int(category) in (1, 2):
            if box in boxes:
                issues["duplicate_person_box"] += 1
                continue
            boxes.append(box)
            attributes.append({"source_class": int(category), "occlusion": int(occlusion), "truncation": int(truncation)})
    return boxes, ignores, attributes, issues


def mask_ignores(image, boxes, ignores):
    result = image.copy()
    for b in ignores:
        x1, y1, x2, y2 = map(int, [np.floor(b[0]), np.floor(b[1]), np.ceil(b[2]), np.ceil(b[3])])
        result[y1:y2, x1:x2] = 114
    # 恢复人体框内像素，避免忽略区与有效GT相交时破坏监督信息。
    for b in boxes:
        x1, y1, x2, y2 = map(int, [np.floor(b[0]), np.floor(b[1]), np.ceil(b[2]), np.ceil(b[3])])
        result[y1:y2, x1:x2] = image[y1:y2, x1:x2]
    return result


def prepare_dataset(raw, output, splits=("train", "val", "test"), limit=None, strict_counts=False):
    raw, output = Path(raw).resolve(), Path(output).resolve()
    if (output / "manifest.json").exists():
        raise FileExistsError(f"已存在数据清单，避免覆盖：{output}。请指定新的输出目录。")
    manifest = {"protocol": "VisDrone2019-DET person=source classes 1+2", "raw_root": str(raw), "splits": {}, "issues": {}}
    hashes, duplicates, global_issues = {}, [], Counter()
    for split in splits:
        src = raw / SPLITS[split]
        images = sorted((src / "images").glob("*.jpg"))
        if not images:
            raise FileNotFoundError(f"未找到 {src / 'images'}")
        if strict_counts and len(images) != EXPECTED_COUNTS[split]:
            raise ValueError(f"{split}应有{EXPECTED_COUNTS[split]}张官方图像，实际{len(images)}张；请核对下载与解压。")
        if limit:
            images = images[:limit]
        target = output / "images" / split
        labels = output / "labels" / split
        originals = output / "originals" / split
        for d in (target, labels, originals):
            d.mkdir(parents=True, exist_ok=True)
        records = []
        for image_path in images:
            digest = sha256(image_path)
            try:
                with Image.open(image_path) as im:
                    image = np.array(im.convert("RGB")); width, height = im.size
            except (OSError, ValueError):
                global_issues["corrupt_image"] += 1
                continue
            ann = src / "annotations" / (image_path.stem + ".txt")
            boxes, ignores, attrs, issues = parse_annotation(ann, width, height)
            annotation_digest = sha256(ann)
            if digest in hashes:
                previous = hashes[digest]
                kind = "within_split" if previous["split"] == split else "across_splits"
                duplicates.append({"first": previous["path"], "second": f"{split}/{image_path.name}",
                                   "kind": kind, "image_sha256": digest,
                                   "same_annotation_file": previous["annotation_sha256"] == annotation_digest})
                global_issues[f"duplicate_{kind}"] += 1
            else:
                hashes[digest] = {"split": split, "path": f"{split}/{image_path.name}",
                                  "annotation_sha256": annotation_digest}
            global_issues.update(issues)
            original_path = originals / image_path.name
            shutil.copy2(image_path, original_path)
            if split == "train" and ignores:
                Image.fromarray(mask_ignores(image, boxes, ignores)).save(target / image_path.name, quality=95)
            else:
                shutil.copy2(image_path, target / image_path.name)
            lines = []
            for x1, y1, x2, y2 in boxes:
                lines.append(f"0 {(x1+x2)/2/width:.8f} {(y1+y2)/2/height:.8f} {(x2-x1)/width:.8f} {(y2-y1)/height:.8f}")
            (labels / (image_path.stem + ".txt")).write_text("\n".join(lines), encoding="utf-8")
            records.append({"id": image_path.stem, "file": image_path.name, "width": width, "height": height,
                            "sha256": digest, "boxes": boxes, "ignores": ignores, "attributes": attrs})
            if len(records) <= 6:
                preview = Image.fromarray(image); draw = ImageDraw.Draw(preview)
                for b in ignores:
                    draw.rectangle(b, outline="orange", width=2)
                for b in boxes:
                    draw.rectangle(b, outline="lime", width=2)
                sample_dir = output / "audit_samples"; sample_dir.mkdir(exist_ok=True)
                preview.save(sample_dir / f"{split}_{image_path.name}")
        manifest["splits"][split] = records
        if strict_counts and len(records) != EXPECTED_COUNTS[split]:
            raise ValueError(f"{split}成功转换{len(records)}张，少于官方{EXPECTED_COUNTS[split]}张；请检查损坏图像。")
    manifest["issues"] = dict(global_issues)
    manifest["duplicate_images"] = duplicates
    manifest["limited_for_smoke_only"] = bool(limit)
    write_json(output / "manifest.json", manifest)
    write_json(output / "duplicate_images.json", duplicates)
    if duplicates:
        print(f"发现{len(duplicates)}对字节相同图像（同划分{global_issues['duplicate_within_split']}，跨划分{global_issues['duplicate_across_splits']}）；已保留官方划分，详见duplicate_images.json。", flush=True)
    cfg = {"path": str(output), "train": "images/train", "val": "images/val", "test": "images/test", "names": {0: "person"}}
    (output / "dataset.yaml").write_text(yaml.safe_dump(cfg, allow_unicode=True, sort_keys=False), encoding="utf-8")
    summarize_dataset(output)
    return manifest


def summarize_dataset(output):
    output = Path(output); manifest = read_json(output / "manifest.json")
    summary, areas, occlusion = {}, [], Counter()
    for split, records in manifest["splits"].items():
        all_boxes = [b for r in records for b in r["boxes"]]
        aa = [(b[2]-b[0])*(b[3]-b[1]) for b in all_boxes]
        summary[split] = {"images": len(records), "negative_images": sum(not r["boxes"] for r in records),
                          "person_boxes": len(aa), "small_area_lt_1024": sum(a < 1024 for a in aa),
                          "ignore_regions": sum(len(r["ignores"]) for r in records)}
        for r in records:
            for b, attr in zip(r["boxes"], r["attributes"]):
                a = (b[2]-b[0])*(b[3]-b[1]); areas.append((split, r["id"], a, b[3]-b[1], attr["occlusion"]))
                occlusion[f"{split}:{attr['occlusion']}"] += 1
    write_json(output / "data_summary.json", {"splits": summary, "occlusion": dict(occlusion), "issues": manifest["issues"],
                                                "duplicate_images": {"within_split": sum(d["kind"] == "within_split" for d in manifest["duplicate_images"]),
                                                                     "across_splits": sum(d["kind"] == "across_splits" for d in manifest["duplicate_images"])}})
    with open(output / "object_sizes.csv", "w", newline="", encoding="utf-8-sig") as f:
        writer = csv.writer(f); writer.writerow(["split", "id", "area_px2", "height_px", "occlusion"]); writer.writerows(areas)
    plot_distributions(output, areas, manifest["issues"])
    return summary


def plot_distributions(output, areas, issues):
    """保存原图像素面积、遮挡程度和原始类别分布，供报告核对。"""
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt

    splits = ["train", "val", "test"]
    values = [np.array([row[2] for row in areas if row[0] == split]) for split in splits]
    fig, ax = plt.subplots(figsize=(7, 4))
    for split, data in zip(splits, values):
        if len(data):
            ax.hist(np.log2(np.maximum(data, 1)), bins=32, alpha=0.45, label=split)
    ax.axvline(10, color="black", linestyle="--", label="32 x 32 px")
    ax.set(xlabel="log2 box area (pixel squared)", ylabel="Person boxes", title="Original-image person area")
    ax.legend(); fig.tight_layout(); fig.savefig(output / "area_distribution.png", dpi=180); plt.close(fig)

    fig, ax = plt.subplots(figsize=(7, 4))
    x = np.arange(3)
    for i, split in enumerate(splits):
        counts = Counter(row[4] for row in areas if row[0] == split)
        ax.bar(x + (i-1)*0.25, [counts.get(level, 0) for level in x], width=0.25, label=split)
    ax.set(xticks=x, xticklabels=["none", "partial", "heavy"], ylabel="Person boxes", title="Occlusion annotations")
    ax.legend(); fig.tight_layout(); fig.savefig(output / "occlusion_distribution.png", dpi=180); plt.close(fig)

    raw = sorted((int(k.removeprefix("raw_class_")), v) for k, v in issues.items() if k.startswith("raw_class_"))
    if raw:
        fig, ax = plt.subplots(figsize=(7, 4))
        ax.bar([str(k) for k, _ in raw], [v for _, v in raw])
        ax.set(xlabel="VisDrone raw category", ylabel="Annotations", title="Raw class distribution")
        fig.tight_layout(); fig.savefig(output / "raw_class_distribution.png", dpi=180); plt.close(fig)

def environment():
    import ultralytics
    import importlib.metadata
    info = {"python": sys.version, "platform": platform.platform(), "torch": torch.__version__,
            "torchvision": importlib.metadata.version("torchvision"),
            "torch_cuda": torch.version.cuda, "ultralytics": ultralytics.__version__, "source": ultralytics.__file__,
            "gpu_available": torch.cuda.is_available(), "created_utc": datetime.now(timezone.utc).isoformat()}
    if info["gpu_available"]:
        info.update(gpu_name=torch.cuda.get_device_name(0), gpu_capability=list(torch.cuda.get_device_capability(0)),
                    gpu_memory_gib=torch.cuda.get_device_properties(0).total_memory/1024**3,
                    compiled_architectures=torch.cuda.get_arch_list())
    try:
        info["nvidia_smi"] = subprocess.run(
            ["nvidia-smi", "--query-gpu=name,memory.total,driver_version", "--format=csv,noheader"],
            capture_output=True, text=True, timeout=10).stdout
    except (OSError, subprocess.TimeoutExpired):
        info["nvidia_smi"] = "unavailable"
    return info


def check_environment(strict=False):
    info = environment(); write_json(ROOT / "outputs/environment.json", info)
    if strict:
        if sys.version_info[:2] != (3, 11) or torch.__version__.split("+")[0] != "2.7.1":
            raise RuntimeError("服务器正式环境要求Python 3.11 / PyTorch 2.7.1，请先运行setup_windows.ps1。")
        if info["torchvision"].split("+")[0] != "0.22.1" or torch.version.cuda != "12.8":
            raise RuntimeError("服务器正式环境要求torchvision 0.22.1与cu128 PyTorch轮子。")
        if not torch.cuda.is_available():
            raise RuntimeError("CUDA不可用，不能启动正式训练。")
        if torch.cuda.get_device_capability(0)[0] >= 12 and tuple(map(int, torch.version.cuda.split(".")[:2])) < (12, 8):
            raise RuntimeError("RTX 5080需要支持Blackwell的CUDA轮子，本项目采用cu128。")
    device = "cuda:0" if torch.cuda.is_available() else "cpu"
    x = torch.randn(16, 16, device=device, requires_grad=True)
    (x @ x.T).sum().backward()
    if not torch.isfinite(x.grad).all():
        raise RuntimeError("CUDA前向/反向校验失败。")
    print(json.dumps(info, ensure_ascii=False, indent=2))
    return info


def get_settings(config):
    settings=yaml.safe_load(Path(config).read_text(encoding="utf-8"))
    required={"epochs","imgsz","batch","seed","optimizer","lr0","amp"}
    if not isinstance(settings,dict) or not required.issubset(settings):
        raise ValueError("--config需要训练设置文件，例如configs/train.yaml；模型结构文件不适用于此参数。")
    return settings


def preflight(data, config, device="0", strict=False):
    """使用最大结构、密集GT和优化器一步检查显存，再统一固定四组batch。"""
    check_environment(strict)
    cfg = get_settings(config); cfg["device"] = device
    manifest = read_json(Path(data).resolve().parent / "manifest.json")
    records = sorted(manifest["splits"]["train"], key=lambda x: len(x["boxes"]), reverse=True)
    if not records:
        raise ValueError("训练集为空")
    target_device = torch.device("cpu" if str(device) == "cpu" else "cuda:" + str(device))
    chosen, peak = None, 0
    for batch_size in (cfg["batch"], max(1, cfg["batch"]//2), max(1, cfg["batch"]//4), 1):
        if chosen is not None:
            break
        model, optimizer, batch, target, loss, scaler = None, None, None, None, None, None
        try:
            if target_device.type == "cuda":
                torch.cuda.empty_cache(); torch.cuda.reset_peak_memory_stats(target_device)
            model = new_model("eca_p2", cfg["seed"]).to(target_device).train()
            model.args = get_cfg(overrides={k:v for k,v in cfg.items() if k != "device"})
            optimizer = torch.optim.AdamW(model.parameters(), lr=cfg["lr0"])
            rows = []
            for i in range(batch_size):
                # 以四块密集标注近似Mosaic的最坏目标数，增强显存探测的保守程度。
                for tile in range(4):
                    rec = records[(i*4+tile) % len(records)]
                    for x1, y1, x2, y2 in rec["boxes"]:
                        rows.append([i, 0, ((x1+x2)/2/rec["width"] + tile%2)/2,
                                     ((y1+y2)/2/rec["height"] + tile//2)/2,
                                     (x2-x1)/rec["width"]/2, (y2-y1)/rec["height"]/2])
            target = torch.tensor(rows, device=target_device, dtype=torch.float32).reshape(-1, 6)
            batch = {"img": torch.rand(batch_size, 3, cfg["imgsz"], cfg["imgsz"], device=target_device),
                     "batch_idx": target[:, 0], "cls": target[:, 1:2], "bboxes": target[:, 2:]}
            amp_enabled = bool(cfg["amp"] and target_device.type == "cuda")
            scaler = torch.amp.GradScaler("cuda", enabled=amp_enabled)
            with torch.autocast(device_type=target_device.type, enabled=amp_enabled):
                loss, _ = model(batch)
            scaler.scale(loss).backward(); scaler.step(optimizer); scaler.update()
            chosen = batch_size
            if target_device.type == "cuda":
                torch.cuda.synchronize(target_device); peak = torch.cuda.max_memory_allocated(target_device)
            print(f"显存检查通过：batch={chosen}, peak={peak/1024**3:.2f} GiB")
        except torch.OutOfMemoryError:
            print(f"batch={batch_size} 显存不足，统一尝试更小batch。")
        finally:
            del model, optimizer, batch, target, loss, scaler
            if target_device.type == "cuda":
                torch.cuda.empty_cache()
    if chosen is None:
        raise RuntimeError("batch=1仍然显存不足，请降低所有实验的输入分辨率后重新检查。")
    cfg["batch"] = chosen
    write_json(ROOT / "outputs/experiment_settings.json", {"settings": cfg, "data": str(Path(data).resolve()),
                "config_sha256": hashlib.sha256(Path(config).read_bytes()).hexdigest(), "probe_peak_bytes": peak,
                "manifest_sha256": hashlib.sha256((Path(data).resolve().parent / "manifest.json").read_bytes()).hexdigest()})
    return cfg


def train_one(variant, data, settings, weights, name=None, resume=None):
    name = name or variant
    overrides = dict(settings, data=str(Path(data).resolve()), project=str(ROOT / "outputs/runs"), name=name,
                     exist_ok=False, pretrained=False)
    if resume:
        overrides["resume"] = str(Path(resume).resolve())
        overrides["model"] = str(Path(resume).resolve())
    else:
        model = new_model(variant, settings["seed"])
        if weights:
            load_backbone(model, weights, ROOT / "outputs/initialization" / f"{name}.json")
        overrides["model"] = str(ROOT / "configs/models" / f"{variant}.yaml")
    trainer = PersonTrainer(overrides=overrides)
    if not resume:
        trainer.model = model
    started = time.perf_counter()
    trainer.train()
    run = trainer.save_dir
    write_json(run / "run_provenance.json", {"variant": variant, "initialization": "shared pretrained backbone only" if weights else "scratch smoke only",
                "settings": settings, "elapsed_seconds": time.perf_counter()-started, "data": str(Path(data).resolve()),
                "data_manifest_sha256": hashlib.sha256((Path(data).resolve().parent / "manifest.json").read_bytes()).hexdigest(),
                "environment": environment(), "resume": str(resume) if resume else None})
    return str(run)


def predict_records(weights, data, split="val", imgsz=960, device="0", half=False):
    root = Path(data).resolve().parent
    annotations = read_json(root / "manifest.json")["splits"][split]
    model = YOLO(str(weights)); records = []
    for i, ann in enumerate(annotations):
        path = root / "originals" / split / ann["file"]
        # ndarray入口避免Windows视频/路径编码差异，坐标仍由Ultralytics还原至原图。
        image = np.array(Image.open(path).convert("RGB"))[:, :, ::-1].copy()
        result = model.predict(image, imgsz=imgsz, conf=0.001, iou=0.7, max_det=1000,
                               device=device, half=half, verbose=False)[0]
        boxes = result.boxes
        pred = torch.cat((boxes.xyxy, boxes.conf[:, None]), 1).cpu().numpy().tolist()
        records.append(dict(ann, pred=pred))
        if (i+1) % 100 == 0:
            print(f"{split}: {i+1}/{len(annotations)}", flush=True)
    return records


def evaluate(weights, data, output, split="val", imgsz=960, device="0", half=False, conf=0.25):
    records = predict_records(weights, data, split, imgsz, device, half)
    out = Path(output); out.mkdir(parents=True, exist_ok=True)
    values = evaluate_records(records, conf)
    values.update(weights=str(Path(weights).resolve()), split=split, imgsz=imgsz, half=half,
                  image_count=len(records), environment=environment())
    write_json(out / "metrics.json", values); write_json(out / "predictions.json", records)
    write_json(out / "threshold_analysis.json", threshold_sweep(records,np.arange(0.05,0.951,0.05)))
    plot_evaluation(records, out)
    create_case_samples(records, Path(data).resolve().parent, split, out)
    return values


def plot_evaluation(records, out):
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt

def benchmark(weights, image, output, imgsz=960, device="0", half=False, warmup=50, repeats=500):
    if repeats < 500 or warmup < 50:
        raise ValueError("正式速度测试要求至少50次预热和500次计时。")
    from ultralytics.data.augment import LetterBox
    from ultralytics.nn.tasks import attempt_load_one_weight
    import cv2
    d = torch.device("cpu" if device == "cpu" else "cuda:" + str(device))
    if half and d.type == "cpu":
        raise ValueError("本项目不在CPU上进行FP16测试。")
    raw = np.array(Image.open(image).convert("RGB"))[:,:,::-1].copy()
    model, _ = attempt_load_one_weight(str(weights), device=d, fuse=True)
    model = model.half() if half else model.float(); model.eval()
    def prep():
        im = LetterBox((imgsz,imgsz), auto=False, stride=32)(image=raw)
        t = torch.from_numpy(np.ascontiguousarray(im[:,:,::-1].transpose(2,0,1))).unsqueeze(0).to(d)
        return (t.half() if half else t.float())/255
    tensor = prep()
    def sync():
        if d.type == "cuda": torch.cuda.synchronize(d)
    times, end_to_end = [], []
    with torch.inference_mode():
        for _ in range(warmup): model(tensor)
        sync()
        if d.type == "cuda": torch.cuda.reset_peak_memory_stats(d)
        for _ in range(repeats):
            sync(); t=time.perf_counter(); model(tensor); sync(); times.append((time.perf_counter()-t)*1000)
        parameters=sum(p.numel() for p in model.parameters())
        model_peak=torch.cuda.max_memory_allocated(d) if d.type=="cuda" else None
        gflops=float(get_flops(model,imgsz))
        del model, tensor
        if d.type=="cuda": torch.cuda.empty_cache()
        predictor=YOLO(str(weights))
        for _ in range(warmup):
            predictor.predict(raw, imgsz=imgsz, rect=False, half=half, device=device, conf=0.25, max_det=1000, verbose=False)
        if d.type=="cuda": torch.cuda.reset_peak_memory_stats(d)
        for _ in range(repeats):
            sync(); t=time.perf_counter()
            predictor.predict(raw, imgsz=imgsz, rect=False, half=half, device=device, conf=0.25, max_det=1000, verbose=False)
            sync(); end_to_end.append((time.perf_counter()-t)*1000)
    result={"imgsz":imgsz,"batch":1,"precision":"FP16" if half else "FP32","warmup":warmup,"repeats":repeats,
            "model_ms_mean":float(np.mean(times)),"model_ms_p95":float(np.percentile(times,95)),
            "end_to_end_ms_mean":float(np.mean(end_to_end)),"end_to_end_ms_p95":float(np.percentile(end_to_end,95)),
            "model_fps":1000/float(np.mean(times)),"end_to_end_fps":1000/float(np.mean(end_to_end)),
            "parameters":parameters,"gflops":gflops,"model_peak_allocated_bytes":model_peak,
            "end_to_end_peak_allocated_bytes":torch.cuda.max_memory_allocated(d) if d.type=="cuda" else None,
            "timing_boundary":"cached BGR image to detections; excludes disk read, plotting and encoding", "environment":environment()}
    write_json(output,result); return result

"""单类、原图坐标、忽略区域感知的101点AP。与官方多类别榜单口径区分。"""
import numpy as np

IOU_THRESHOLDS = np.linspace(0.5, 0.95, 10)


def overlap(a, b, ioa=False):
    a, b = np.asarray(a, dtype=float).reshape(-1, 4), np.asarray(b, dtype=float).reshape(-1, 4)
    inter = np.maximum(np.minimum(a[:, None, 2:], b[None, :, 2:]) -
                       np.maximum(a[:, None, :2], b[None, :, :2]), 0).prod(2)
    aa = np.maximum(a[:, 2:] - a[:, :2], 0).prod(1)[:, None]
    bb = np.maximum(b[:, 2:] - b[:, :2], 0).prod(1)[None, :]
    return inter / np.maximum(aa if ioa else aa + bb - inter, 1e-12)


def match_image(pred, boxes, ignores, iou=0.5, small=False):
    """先匹配计分GT，再忽略非计分GT/区域；重复检测仍可能是FP。"""
    pred = np.asarray(pred, dtype=float).reshape(-1, 5)
    boxes = np.asarray(boxes, dtype=float).reshape(-1, 4)
    ignores = np.asarray(ignores, dtype=float).reshape(-1, 4)
    areas = np.maximum(boxes[:, 2:] - boxes[:, :2], 0).prod(1)
    active = areas < 1024 if small else np.ones(len(boxes), dtype=bool)
    gt, non_scoring = boxes[active], boxes[~active]
    pred = pred[np.argsort(-pred[:, 4], kind="stable")]
    used = set()
    tp, fp, scores, ignored = [], [], [], 0
    valid_iou = overlap(pred[:, :4], gt)
    other_iou = overlap(pred[:, :4], non_scoring)
    ignore_ioa = overlap(pred[:, :4], ignores, ioa=True)
    for j, p in enumerate(pred):
        candidates = [g for g in range(len(gt)) if g not in used and valid_iou[j, g] >= iou]
        if candidates:
            g = max(candidates, key=lambda g: valid_iou[j, g])
            used.add(g)
            tp.append(1); fp.append(0); scores.append(p[4])
        elif ((len(non_scoring) and other_iou[j].max() >= iou) or
              (len(ignores) and ignore_ioa[j].max() >= 0.5)):
            ignored += 1
        else:
            tp.append(0); fp.append(1); scores.append(p[4])
    return np.asarray(scores), np.asarray(tp), np.asarray(fp), len(gt), ignored


def curve(records, iou=0.5, small=False):
    items = [match_image(r["pred"], r["boxes"], r.get("ignores", []), iou, small) for r in records]
    n = sum(x[3] for x in items)
    scores = np.concatenate([x[0] for x in items]) if items else np.array([])
    tp = np.concatenate([x[1] for x in items]) if items else np.array([])
    fp = np.concatenate([x[2] for x in items]) if items else np.array([])
    order = np.argsort(-scores, kind="stable")
    scores, tp, fp = scores[order], tp[order], fp[order]
    recall = np.cumsum(tp) / max(n, 1)
    precision = np.cumsum(tp) / np.maximum(np.cumsum(tp + fp), 1)
    # 在101个召回位置取得右侧最大精度；没有预测时AP为0，没有GT时不定义。
    ap = float(np.mean([precision[recall >= r].max() if np.any(recall >= r) else 0.0
                        for r in np.linspace(0, 1, 101)])) if n else None
    return {"ap": ap, "scores": scores, "tp": tp, "fp": fp, "recall": recall,
            "precision": precision, "gt": n, "ignored": sum(x[4] for x in items)}


def evaluate_records(records, conf=0.25):
    aps = [curve(records, float(t))["ap"] for t in IOU_THRESHOLDS]
    small_aps = [curve(records, float(t), True)["ap"] for t in IOU_THRESHOLDS]
    c = curve(records)
    selected = c["scores"] >= conf
    tp, fp = int(c["tp"][selected].sum()), int(c["fp"][selected].sum())
    p, r = tp / max(tp + fp, 1), tp / max(c["gt"], 1)
    return {"precision": p, "recall": r, "f1": 2*p*r/max(p+r, 1e-12),
            "map50": aps[0], "map50_95": float(np.mean(aps)) if aps[0] is not None else None,
            "ap_small": float(np.mean(small_aps)) if small_aps[0] is not None else None,
            "ap_by_iou": aps, "tp": tp, "fp": fp, "fn": c["gt"]-tp,
            "ground_truth_count": c["gt"], "ignored_predictions_at_iou50": c["ignored"],
            "operating_confidence": conf, "protocol": "person-only; 101 recall points; max_det=1000; native-image area<1024 for small"}


def threshold_sweep(records, thresholds):
    """阈值变化不影响完整AP；一次匹配即可计算所有P/R工作点。"""
    c=curve(records)
    rows=[]
    for threshold in thresholds:
        keep=c["scores"]>=threshold
        tp,fp=int(c["tp"][keep].sum()),int(c["fp"][keep].sum())
        p,r=tp/max(tp+fp,1),tp/max(c["gt"],1)
        rows.append({"operating_confidence":float(threshold),"precision":p,"recall":r,
                     "f1":2*p*r/max(p+r,1e-12),"tp":tp,"fp":fp,"fn":c["gt"]-tp})
    return rows

"""C2f 的通道注意力扩展；保留 cv1/cv2/m 名称以加载相同预训练骨干。"""
from . import runtime
import torch
from torch import nn
from ultralytics.nn.modules.block import C2f


class ECA(nn.Module):
    def __init__(self, kernel_size=3):
        super().__init__()
        self.pool = nn.AdaptiveAvgPool2d(1)
        self.conv = nn.Conv1d(1, 1, kernel_size, padding=kernel_size // 2, bias=False)

    def forward(self, x):
        # B,C,H,W -> B,1,C；只沿通道维进行局部交互。
        y = self.pool(x).squeeze(-1).transpose(1, 2)
        y = self.conv(y).transpose(1, 2).unsqueeze(-1).sigmoid()
        return x * y


class C2fECA(C2f):
    def __init__(self, c1, c2, n=1, shortcut=False, g=1, e=0.5):
        super().__init__(c1, c2, n, shortcut, g, e)
        self.eca = ECA(3)

    def forward(self, x):
        return self.eca(super().forward(x))

    def forward_split(self, x):
        return self.eca(super().forward_split(x))

class PersonValidator(DetectionValidator):
    def init_metrics(self, model):
        super().init_metrics(model)
        manifest = read_json(Path(self.data["path"]) / "manifest.json")
        self.annotation_index = {(split,r["id"]): r for split,records in manifest["splits"].items() for r in records}
        self.records = []

    def update_metrics(self, preds, batch):
        for si, pred in enumerate(preds):
            self.seen += 1
            prepared = self._prepare_batch(si, batch)
            native = self._prepare_pred(pred, prepared).detach().cpu().numpy()
            image_path=Path(batch["im_file"][si])
            rid = image_path.stem
            ann = self.annotation_index[(image_path.parent.name,rid)]
            self.records.append({"id": rid, "pred": native[:, :5].tolist(), "boxes": ann["boxes"], "ignores": ann["ignores"]})

    def get_stats(self):
        values = evaluate_records(self.records)
        if not values["ground_truth_count"]:
            raise ValueError("验证集没有有效人体GT，无法选择最佳模型。")
        p, r, f1 = values["precision"], values["recall"], values["f1"]
        x = np.linspace(0, 1, 101)
        c = curve(self.records)
        pr = np.array([c["precision"][c["recall"] >= t].max() if np.any(c["recall"] >= t) else 0 for t in x])
        pc, rc = [], []
        for threshold in x:
            selected = c["scores"] >= threshold
            tp, fp = c["tp"][selected].sum(), c["fp"][selected].sum()
            pc.append(tp/max(tp+fp, 1)); rc.append(tp/max(c["gt"], 1))
        pc, rc = np.array([pc]), np.array([rc])
        self.metrics.box.nc = 1
        self.metrics.box.update((np.array([p]), np.array([r]), np.array([f1]),

        self.nt_per_image = np.array([sum(bool(r["boxes"]) for r in self.records)])
        write_json(self.save_dir / "person_metrics.json", values)

    def get_validator(self):
        self.loss_names = "box_loss", "cls_loss", "dfl_loss"
        return PersonValidator(self.test_loader, save_dir=self.save_dir, args=copy(self.args), _callbacks=self.callbacks)

    def get_dataloader(self, dataset_path, batch_size=16, rank=0, mode="train"):
        try:
            return super().get_dataloader(dataset_path, batch_size, rank, mode)
        except (OSError, RuntimeError) as exc:
            if self.args.workers == 0:
                raise
            print(f"数据加载进程创建失败，回退workers=0: {exc}")
            self.args.workers = 0
            return super().get_dataloader(dataset_path, batch_size, rank, mode)

        elif args.command=="pack": pack_results(args.output,not args.no_weights)
        elif args.command=="summary": print(summarize_results())


if __name__=="__main__":
    multiprocessing.freeze_support()
    main()
